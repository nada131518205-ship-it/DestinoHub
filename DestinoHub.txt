-- RightShift mostra/esconde | "–" minimiza pra pílula | "X" descarrega tudo

local Workspace     = game:GetService("Workspace")
local RunService    = game:GetService("RunService")
local Players       = game:GetService("Players")
local UIS           = game:GetService("UserInputService")
local RS            = game:GetService("ReplicatedStorage")
local VIM           = game:GetService("VirtualInputManager")
local TweenService  = game:GetService("TweenService")
local lp            = Players.LocalPlayer

if getgenv().DestinoHub then pcall(getgenv().DestinoHub.Unload) end

local Hub = {alive = true, notify = function() end}
local conns, drawings = {}, {}

local function bind(sig, fn)
    local c = sig:Connect(fn)
    conns[#conns + 1] = c
    return c
end

local Cfg = {
    Parry     = true,
    Overlay   = true,
    Debug     = false,
    Dash      = false,
    Raio      = 25,
    Lead      = 0.12,
    ClickDist = 10,
    ClickMax  = 40,
    Smooth    = 0.5,
    Delay     = 4,
}

local Parry, Items, Dash = {}, {}, {}

-- ════════════════════════ PARRY ════════════════════════
do
    local MESH_ID    = "rbxassetid://15100461919"
    local MARGEM     = 3
    local LOCK_GATE  = 12
    local MIN_MOVE   = 0.2
    local GATE, GATE_MISS = 16, 3
    local PERDA      = 90
    local CLICK_REARM, CLICK_CD = 4, 0.08

    local EV_NAME, EV_WINDOW, EV_RESCAN, EV_RECHECK = "Event", 0.35, 0.5, 0.25

    local cam = Workspace.CurrentCamera

    local function acharAlvo()
        for _, o in ipairs(Workspace:GetDescendants()) do
            if o:IsA("MeshPart") and o.MeshId == MESH_ID then return o end
        end
    end

    local alvo = acharAlvo()
    local aparecido = os.clock()

    local frame = 0
    local buf, prev = {}, {}
    local pos, vel, ultFrame = nil, Vector3.zero, 0
    local aceitas, rejeitadas, maxN = 0, 0, 0

    local oldNC; oldNC = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        if Hub.alive and Cfg.Parry and alvo and getnamecallmethod() == "BulkMoveTo" and not checkcaller() then
            local partes, cfs = ...
            if type(partes) == "table" and type(cfs) == "table" then
                for i = 1, #partes do
                    if partes[i] == alvo then
                        local cf = cfs[i]
                        if typeof(cf) == "CFrame" then buf[#buf + 1] = cf.Position end
                    end
                end
            end
        end
        return oldNC(self, ...)
    end))

    local function rootPos()
        local r = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        return r and r.Position
    end

    local function adquirir()
        local rp = rootPos()
        local best, bd = nil, math.huge
        for _, p in ipairs(buf) do
            for _, q in ipairs(prev) do
                local m = (p - q).Magnitude
                if m >= MIN_MOVE and m <= LOCK_GATE then
                    local d = rp and (p - rp).Magnitude or 0
                    if d < bd then best, bd = p, d end
                end
            end
        end
        if best then
            pos, vel, ultFrame = best, Vector3.zero, frame
            aceitas += 1
        end
    end

    local function seguir()
        local df = frame - ultFrame
        local previsto = pos + vel * df
        local best, bd = nil, math.huge
        for _, p in ipairs(buf) do
            local d = (p - previsto).Magnitude
            if d < bd then best, bd = p, d end
        end
        if best and bd <= GATE + GATE_MISS * math.max(df, 1) then
            if df > 0 then vel = vel:Lerp((best - pos) / df, Cfg.Smooth) end
            pos, ultFrame = best, frame
            aceitas += 1
        else
            rejeitadas += 1
        end
    end

    local function atualPos()
        if not pos then return nil end
        return pos + vel * math.min(frame - ultFrame, 10)
    end

    bind(UIS.InputBegan, function(i, gp)
        if not gp and i.KeyCode == Enum.KeyCode.R then
            pos = nil
            Hub.notify("relock")
        end
    end)

    -- rastreio
    local dentro, busca, tlog = false, 0, 0
    bind(RunService.Heartbeat, function(dt)
        if not Cfg.Parry then
            if #buf > 0 or #prev > 0 then table.clear(buf) table.clear(prev) end
            pos = nil
            return
        end

        frame += 1
        maxN = math.max(maxN, #buf)

        if not alvo or not alvo:IsDescendantOf(Workspace) then
            alvo, pos = nil, nil
            table.clear(buf) table.clear(prev)
            busca += dt
            if busca >= 0.25 then
                busca = 0
                alvo = acharAlvo()
                if alvo then aparecido = os.clock() end
            end
            if dentro then dentro = false end
            return
        end

        if os.clock() - aparecido < Cfg.Delay then
            table.clear(buf) table.clear(prev)
            pos = nil
            return
        end

        if #buf > 0 then
            if pos then seguir() else adquirir() end
            prev, buf = buf, {}
        end

        if pos and frame - ultFrame > PERDA then pos = nil end

        tlog += dt
        if Cfg.Debug and tlog >= 2 then
            tlog = 0
            print(("[track] lock=%s aceitas=%d rej=%d maxamostras=%d pos=%s"):format(
                tostring(pos ~= nil), aceitas, rejeitadas, maxN, tostring(pos)))
        end

        local rp, atual = rootPos(), atualPos()
        if not atual or not rp then return end
        local d = (rp - atual).Magnitude
        if not dentro and d <= Cfg.Raio then
            dentro = true
            if Cfg.Debug then print(("entrou %.1f"):format(d)) end
        elseif dentro and d > Cfg.Raio + MARGEM then
            dentro = false
            if Cfg.Debug then print(("saiu %.1f"):format(d)) end
        end
    end)

    -- overlay
    local dot = Drawing.new("Circle")
    dot.Radius, dot.Thickness, dot.Filled = 9, 2, false
    local txt = Drawing.new("Text")
    txt.Size, txt.Center, txt.Outline, txt.Color = 14, true, true, Color3.new(1, 1, 1)
    local line = Drawing.new("Line")
    line.Thickness, line.Color = 1, Color3.fromRGB(255, 60, 60)
    drawings[#drawings + 1] = dot
    drawings[#drawings + 1] = txt
    drawings[#drawings + 1] = line

    bind(RunService.RenderStepped, function()
        cam = Workspace.CurrentCamera or cam
        local atual = atualPos()
        if not (Cfg.Parry and Cfg.Overlay) or not atual then
            dot.Visible, txt.Visible, line.Visible = false, false, false
            return
        end

        local sp, vis = cam:WorldToViewportPoint(atual)
        dot.Visible, txt.Visible = vis, vis
        if not vis then line.Visible = false return end

        local p2 = Vector2.new(sp.X, sp.Y)
        local rp = rootPos()
        local d = rp and (rp - atual).Magnitude or 0
        dot.Position, txt.Position = p2, p2 + Vector2.new(0, 16)
        dot.Color = d <= Cfg.Raio and Color3.fromRGB(255, 220, 60) or Color3.fromRGB(255, 60, 60)
        txt.Text = ("%.0f"):format(d)

        local fp = cam:WorldToViewportPoint(atual + vel * 15)
        line.From, line.To, line.Visible = p2, Vector2.new(fp.X, fp.Y), true
    end)

    -- autoclick / Event
    local Event, evOk, evChecked = nil, false, 0
    local windowUntil, lastFire, settling = 0, 0, false
    local pending  = setmetatable({}, {__mode = "k"})
    local listened = setmetatable({}, {__mode = "k"})

    local function idnum(obj)
        return tonumber(obj:GetDebugId():match("_(%d+)")) or 0
    end

    local function isEvent(obj)
        return obj.Name == EV_NAME and obj:IsA("BindableEvent")
    end

    local function listen(obj)
        if listened[obj] then return end
        listened[obj] = true
        bind(obj.Event, function(...)
            local now = os.clock()
            if now > windowUntil then return end
            if now - lastFire < 0.1 then return end
            if select("#", ...) ~= 0 then return end
            pending[obj] = true
        end)
    end

    local function scan()
        for _, obj in getnilinstances() do
            if isEvent(obj) then listen(obj) end
        end
    end

    task.spawn(function()
        while Hub.alive do
            scan()
            task.wait(EV_RESCAN)
        end
    end)

    local function settle()
        settling = false
        local best, bestId
        for obj in pending do
            local n = idnum(obj)
            if not bestId or n < bestId then best, bestId = obj, n end
        end
        table.clear(pending)
        if best then
            Event, evOk, evChecked = best, true, os.clock()
            Hub.notify("Event capturado")
        end
    end

    local function openWindow()
        if settling then return end
        settling = true
        windowUntil = os.clock() + EV_WINDOW
        scan()
        task.delay(EV_WINDOW + 0.05, settle)
    end

    local function evAlive(now)
        if not Event then return false end
        if now - evChecked >= EV_RECHECK then
            evChecked = now
            local ok, c = pcall(getconnections, Event.Event)
            evOk = ok and #c > 0
        end
        return evOk
    end

    local function cliqueReal()
        if mouse1click then
            mouse1click()
        else
            local m = UIS:GetMouseLocation()
            VIM:SendMouseButtonEvent(m.X, m.Y, 0, true, game, 0)
            VIM:SendMouseButtonEvent(m.X, m.Y, 0, false, game, 0)
        end
    end

    local function aprender(espera)
        task.wait(espera)
        for _ = 1, 3 do
            if not Hub.alive or evAlive(os.clock()) then break end
            openWindow()
            task.wait(0.05)
            cliqueReal()
            task.wait(EV_WINDOW + 0.3)
        end
    end
    task.spawn(aprender, 1)

    local function disparar()
        local now = os.clock()
        if not evAlive(now) then return false end
        lastFire = now
        if pcall(Event.Fire, Event) then return true end
        evOk = false
        return false
    end

    bind(UIS.InputBegan, function(input, gpe)
        if gpe then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 then openWindow() end
    end)

    local function ehVermelho(c)
        return c.R > 0.6 and c.G < 0.35 and c.B < 0.35
    end

    local function bolaVermelha()
        if not alvo then return false end
        if ehVermelho(alvo.Color) then return true end
        for _, o in ipairs(alvo:GetChildren()) do
            if o:IsA("Highlight") and (ehVermelho(o.FillColor) or ehVermelho(o.OutlineColor)) then
                return true
            end
        end
        return false
    end

    local armado, ultClick, tdbg = true, 0, 0

    bind(RunService.Heartbeat, function(dt)
        if not Cfg.Parry then armado = true return end
        local rp, atual = rootPos(), atualPos()
        if not alvo or not rp or not atual then armado = true return end

        local agora    = os.clock()
        local d        = (rp - atual).Magnitude
        local vermelha = bolaVermelha()

        if not vermelha then armado = true end

        local vs = vel.Magnitude / math.max(dt, 1 / 240)
        local gatilho = vs * Cfg.Lead
        if gatilho ~= gatilho or gatilho == math.huge then gatilho = Cfg.ClickDist end
        gatilho = math.clamp(gatilho, Cfg.ClickDist, Cfg.ClickMax)

        if vermelha and d <= gatilho and armado and agora - ultClick >= CLICK_CD then
            if disparar() then armado, ultClick = false, agora end
        elseif vermelha and d > gatilho + CLICK_REARM then
            armado = true
        end

        if Cfg.Debug and vermelha and d <= gatilho + 5 and agora - tdbg > 1 then
            tdbg = agora
            print(("[parry] d=%.1f gat=%.1f armado=%s event=%s"):format(
                d, gatilho, tostring(armado), tostring(evAlive(agora))))
        end
    end)

    function Parry.Relock() pos = nil Hub.notify("relock") end
    function Parry.Relearn()
        Event, evOk = nil, false
        Hub.notify("aprendendo Event...")
        task.spawn(aprender, 0)
    end
    function Parry.Stats()
        local rp, atual = rootPos(), atualPos()
        local d = (rp and atual) and (rp - atual).Magnitude or nil
        return {
            lock = pos ~= nil, aceitas = aceitas, rej = rejeitadas, dist = d,
            alvo = alvo ~= nil, event = evAlive(os.clock()),
        }
    end
end

-- ════════════════════════ DASH ════════════════════════
do
    function Dash.Scan()
        local n = 0
        for _, v in next, getgc(true) do
            if type(v) == "table" then
                for k, val in next, v do
                    if type(k) == "string" and k:lower():find("cooldown") and type(val) == "number" then
                        print("[cd]", k, val, v)
                        n += 1
                    end
                end
            end
        end
        for k, v in next, lp:GetAttributes() do
            if tostring(k):lower():find("cooldown") then print("[attr]", k, v) n += 1 end
        end
        return n
    end

    function Dash.Patch()
        local n = 0
        for _, t in next, getgc(true) do
            if type(t) == "table" then
                if rawget(t, "DASH_COOLDOWN_TIME") ~= nil then
                    rawset(t, "DASH_COOLDOWN_TIME", 0)
                    rawset(t, "DEFLECT_COOLDOWN_TIME", 0)
                    n += 1
                end
                local d = rawget(t, "DashCooldown")
                if d == 0.7 then rawset(t, "DashCooldown", 0) n += 1 end
                local c = rawget(t, "CooldownDuration")
                if c == 0.7 or c == 1.3 then rawset(t, "CooldownDuration", 0) n += 1 end
            end
        end
        return n
    end

    task.spawn(function()
        while Hub.alive do
            task.wait(5)
            if Cfg.Dash then Dash.lastN = Dash.Patch() end
        end
    end)
end

local function toggle(par, label, key, cb)
    local row = mk("TextButton", {Size = UDim2.new(1, -6, 0, 36), BackgroundColor3 = T.item, Text = "",
        AutoButtonColor = false}, par)
    corner(row)
    mk("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(12, 0), Size = UDim2.new(1, -70, 1, 0),
        Text = label, TextColor3 = T.text, Font = Enum.Font.Gotham, TextSize = 14,
        TextXAlignment = Enum.TextXAlignment.Left}, row)
    local sw = mk("Frame", {Size = UDim2.fromOffset(40, 20), Position = UDim2.new(1, -52, 0.5, -10),
        BackgroundColor3 = T.panel}, row)
    corner(sw, 10)
    local knob = mk("Frame", {Size = UDim2.fromOffset(16, 16), Position = UDim2.fromOffset(2, 2),
        BackgroundColor3 = T.text}, sw)
    corner(knob, 8)
    local function paint()
        local on = Cfg[key]
        tween(sw, {BackgroundColor3 = on and T.accent or T.panel})
        tween(knob, {Position = on and UDim2.fromOffset(22, 2) or UDim2.fromOffset(2, 2)})
    end
    row.MouseButton1Click:Connect(function()
        Cfg[key] = not Cfg[key]
        paint()
        if cb then cb(Cfg[key]) end
    end)
    paint()
end

local function slider(par, label, key, min, max, step, fmt)
    local row = mk("Frame", {Size = UDim2.new(1, -6, 0, 46), BackgroundColor3 = T.item}, par)
    corner(row)
    local lab = mk("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(12, 4),
        Size = UDim2.new(1, -24, 0, 18), Text = "", TextColor3 = T.text, Font = Enum.Font.Gotham, TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left}, row)
    local track = mk("Frame", {Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 0, 32),
        BackgroundColor3 = T.panel}, row)
    corner(track, 3)
    local fill = mk("Frame", {Size = UDim2.fromScale(0, 1), BackgroundColor3 = T.accent}, track)
    corner(fill, 3)
    mk("UIGradient", {Color = ColorSequence.new(T.accent, T.accent2)}, fill)

    local function refresh()
        fill.Size = UDim2.fromScale(math.clamp((Cfg[key] - min) / (max - min), 0, 1), 1)
        lab.Text = label .. ":  " .. string.format(fmt, Cfg[key])
    end
    local dragging = false
    local function set(x)
        local a = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        Cfg[key] = math.floor((min + a * (max - min)) / step + 0.5) * step
        refresh()
    end
    row.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            set(i.Position.X)
        end
    end)
    bind(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            set(i.Position.X)
        end
    end)
    bind(UIS.InputEnded, function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    refresh()
end

local function renderItens()

    equipBtn.BackgroundColor3 = Items.equipado() and Color3.fromRGB(0, 140, 70) or T.item
    tE.BackgroundColor3 = sub == "Espadas" and T.accent or T.item
    tP.BackgroundColor3 = sub == "Poses" and T.accent or T.item

    for _, c in ipairs(lista:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
    local q = busca.Text:lower()
    for _, item in ipairs(sub == "Espadas" and Items.ESPADAS or Items.POSES) do
        local nome, id = item, nil
        if type(item) == "table" then nome, id = item[1], item[2] end
        if q == "" or nome:lower():find(q, 1, true) then
            local sel = id and id == Items.poseAtual() or (not id and nome == Items.espadaAtual())
            local b = mk("TextButton", {Size = UDim2.new(1, -6, 0, 26), BackgroundColor3 = sel and T.accent or T.item,
                TextColor3 = T.text, Text = "  " .. nome, TextXAlignment = Enum.TextXAlignment.Left,
                Font = Enum.Font.Gotham, TextSize = 13, TextTruncate = Enum.TextTruncate.AtEnd}, lista)
            corner(b, 6)
            b.MouseButton1Click:Connect(function()
                if id then Items.SetPose(id) else Items.SetSword(nome) end
            end)
        end
    end
end

function Hub.Unload()
    Hub.alive = false
    for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    for _, d in ipairs(drawings) do pcall(function() d:Remove() end) end
    pcall(Items.Unequip)
    pcall(function() gui:Destroy() end)
    getgenv().DestinoHub = nil
end
